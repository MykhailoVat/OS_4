# demo.txt
# Усі обов'язкові команди варіанта ФС у пам'яті.
create notes
stat notes
ls
open notes
write 0 5 hello
seek 0 0
read 0 5
link notes copy
stat copy
ls
# Незалежне зміщення другого відкриття.
open copy
read 1 2
# Розширення не виділяє нульові блоки.
truncate copy 1024
stat notes
seek 0 510
write 0 4 DATA
seek 0 508
read 0 8
# Відкинутий хвіст не повертається після збільшення.
truncate notes 3
truncate notes 8
seek 0 0
read 0 8
# Файл доступний через fd навіть після видалення обох імен.
unlink notes
unlink copy
ls
seek 0 0
read 0 8
write 0 1 !
close 0
# Найменший вільний fd знову 0; старий файл ще відкритий через fd=1.
create fresh
open fresh
close 1
close 0
unlink fresh
ls
stat /
exit
